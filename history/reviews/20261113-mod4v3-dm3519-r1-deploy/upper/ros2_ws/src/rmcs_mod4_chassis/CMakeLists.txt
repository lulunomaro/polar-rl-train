cmake_minimum_required(VERSION 3.16)
project(rmcs_mod4_chassis LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(MOD4_WITH_ROS "Build ROS2 standalone node and RMCS plugin" ON)
add_library(mod4_transport STATIC src/bridge.cpp src/mod4_wire.c)
target_include_directories(mod4_transport PUBLIC include)
target_compile_options(mod4_transport PRIVATE -Wall -Wextra -Wpedantic)
if(MOD4_WITH_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(pluginlib REQUIRED)
  find_package(rmcs_executor REQUIRED)
  add_library(mod4_chassis_component SHARED src/component.cpp)
  target_link_libraries(mod4_chassis_component mod4_transport)
  ament_target_dependencies(mod4_chassis_component rclcpp geometry_msgs std_msgs pluginlib rmcs_executor)
  add_executable(mod4_chassis_node src/main.cpp)
  target_link_libraries(mod4_chassis_node mod4_transport)
  ament_target_dependencies(mod4_chassis_node rclcpp geometry_msgs std_msgs)
  pluginlib_export_plugin_description_file(rmcs_executor plugins.xml)
  install(TARGETS mod4_chassis_component DESTINATION lib)
  install(TARGETS mod4_chassis_node DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
  ament_package()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(mod4_transport_test tests/transport_test.cpp)
  target_link_libraries(mod4_transport_test mod4_transport util)
  target_compile_options(mod4_transport_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME mod4_transport_pty COMMAND mod4_transport_test)
endif()
