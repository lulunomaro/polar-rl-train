    if not bool(((mask == 0) | (mask == 1)).all()):
        raise ValueError("binary native episode-prefix mask required")
    if not bool((mask[:, 0] == 1).all()) or bool((mask[:, 1:] > mask[:, :-1]).any()):
        raise ValueError("mask must retain first/terminal and exclude only suffix")
    actual = target.detach()
    error = ((predicted - actual) / scale_m).square().mean(dim=(-1, -2))
    weight = mask.to(error.dtype)
    objective = (error * weight).sum() / predicted.shape[0]
    valid = weight.sum()

    return {
        "vector_objective_sumtime_perbatch": objective,
        "vector_mean_per_valid_transition": (error * weight).sum() / valid,
        "vector_error_per_transition": error,
        "predicted_vectors_body_m": predicted,
        "actual_vectors_body_m": actual,

    }


class OwnClosureVectorSupervision(nn.Module):
    """Reuse pinned own FK. No learned head and no ideal-loop reconstruction.

    Canonical chassis p=0/R=I prevents world translation/heading and large-world
    cancellation from contaminating closure-vector labels. This is only a frame
    choice for evaluating FK; it does not write any plant or predicted state.
    q30 is passed unchanged. dq/aux are unused pose inputs to the existing FK.
    """
    def __init__(self, geometry: nn.Module, contract: dict):
        super().__init__()
        self.geometry = geometry
        body_names = tuple(geometry.body_names)
        rows = tuple(contract["closures"].values())
        if len(rows) != 6 or len(body_names) != 31:
            raise ValueError("pinned own31body/sixclosure contract required")
        if any(True for _ in geometry.parameters()):
            raise ValueError("source FK must not contain a learned predictor")
        self.register_buffer("i0", torch.tensor([body_names.index(r["body0"]) for r in rows]))
        self.register_buffer("i1", torch.tensor([body_names.index(r["body1"]) for r in rows]))
        self.register_buffer("a0", torch.tensor([r["anchor0_m"] for r in rows], dtype=torch.float32))
        self.register_buffer("a1", torch.tensor([r["anchor1_m"] for r in rows], dtype=torch.float32))

    def vectors(self, state):
        if len(state) != 7 or state[4].shape[-1] != 30:
            raise ValueError("native sevencomponent/physical30 state required")
        p, _, v, w, q, dq, aux = state
        R0 = torch.eye(3, dtype=q.dtype, device=q.device).expand(*q.shape[:-1], 3, 3)
        canonical = (torch.zeros_like(p), R0, torch.zeros_like(v), torch.zeros_like(w),
                     q, torch.zeros_like(dq), torch.zeros_like(aux))
        g = self.geometry(canonical)
        P, R = g["body_position"], g["body_rotation"]
        a = P[..., self.i0, :] + (R[..., self.i0, :, :] @ self.a0[..., None]).squeeze(-1)
        b = P[..., self.i1, :] + (R[..., self.i1, :, :] @ self.a1[..., None]).squeeze(-1)
        return a - b

    def forward(self, predicted_state, actual_state, mask, *, scale_m: float):
        predicted = self.vectors(predicted_state)
        with torch.no_grad():
            target = self.vectors(actual_state)
        return labelled_vector_loss(predicted, target, mask, scale_m=scale_m)


def combine_candidate_b_only(base_objective, vector_terms, *, explicit_vector_weight: float):
